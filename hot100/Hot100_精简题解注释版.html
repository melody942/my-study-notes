5.1 固定大小堆维护 Top K

215 数组中的第 K 个最大元素

力扣原题 ↗
题目简述

215 数组中的第 K 个最大元素：给定无序数组和 K，返回按排序名次计算的第 K 大元素，而非第 K 个不同值。

题解代码C++展开
// 关键注释（来自个人笔记）
// 题目理解：
// - 实则堆的时间复杂度超过题目要求，还要学会快速选择算法，理解起来也简单
// - 随机选一个划分点，rand()%n，遍历一遍，大于它的进big数组，小于的进small数组，等于的进equal数组
// - 如果k小于big数组的长度，说明答案在big里面，return(big，k)
// - 如果k大于big+equal数组的长度，说明答案在small里面，return(small,k-big-equal)为什么k也要变？因为新的 k = 原来的 k - (被我们淘汰掉的比他强的人数)
// - 再不然划分点即答案。
// 思路：
// - 关键是学会使用堆
// - 小顶堆：堆顶是最小值，用来找第k个最大值 priorityqueue,greater> pq;
// - 大顶堆：堆顶是最大值，用来找第k个最小值 priorityqueue pq;
class Solution {
public:
    int findKthLargest(vector<int>& nums, int k) {
        priority_queue<int,vector<int>,greater<int>> pq;
        for(int x:nums){
            pq.push(x);
            if(pq.size()>k){
                pq.pop();
            }
        }
        return pq.top();
    }
};

class Solution {
public:
    int quick(vector<int>&nums,int k){
        int pivot=nums[rand()%nums.size()];
        vector<int> small,equal,big;
        for(int num:nums){
            if(num>pivot) big.push_back(num);
            else if(num<pivot) small.push_back(num);
            else equal.push_back(num);
        }
        if(k<=big.size()) return quick(big,k);
        if(k>(big.size()+equal.size())) return quick(small,k-big.size()-equal.size());
        return pivot;
    }

    int findKthLargest(vector<int>& nums, int k) {
        return quick(nums,k);
    }
};

5.1 固定大小堆维护 Top K

模板：共同骨架

题目简述

第 K 小变形：给定整数数组和 K，返回按排序名次计算的第 K 小元素。

题解代码C++展开
// 关键注释
// 核心思路：维护大小为 K 的大顶堆，堆中始终保留当前最小的 K 个元素。
// 关键关注：堆顶是这 K 个元素中最大的边界，遍历结束后它就是第 K 小。
// 易错提醒：只有堆大小超过 K 时才弹出堆顶，不能提前丢失候选。
int kthSmallestTemplate(const vector<int>& nums, int k) {
    priority_queue<int> heap; // 大顶堆保留当前最小的 K 个元素。

    for (int x : nums) {
        heap.push(x);
        if ((int)heap.size() > k) {
            heap.pop(); // 淘汰这 K + 1 个候选中最大的元素。
        }
    }

    return heap.top();
}

import heapq


def kthSmallestTemplate(nums, k):
    heap = []
    for x in nums:
        # 关键步骤：维护大小为 K 的大顶堆，堆中始终保留当前最小的 K 个元素。
        heapq.heappush(heap, -x)
        if len(heap) > k:
            heapq.heappop(heap)
    # 关键不变量：堆顶是这 K 个元素中最大的边界，遍历结束后它就是第 K 小。
    return -heap[0]

5.2 哈希计数 + 固定大小堆

347 前 K 个高频元素

力扣原题 ↗
题目简述

347 前 K 个高频元素：给定整数数组和 K，返回出现频率最高的 K 个元素，结果顺序不限。

题解代码C++展开
// 关键注释（来自个人笔记）
// 题目理解：
// - vector> buckets(n + 1);
// - for (auto& [num, freq] : count) {
// - buckets[freq].pushback(num);
// - }
// 思路：
// - 学习一个叫做桶排序的东西。
// - 首先遍历数组做哈希表，数字和次数。
// - 桶排序就是，把出现次数相同的数字，放入同一个数组中
// - 然后从桶的n开始遍历，把桶里的数字放入res，直到k个
// - 堆排序时间复杂度超过了/
class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        unordered_map<int, int> count;
        for (int num : nums) {
            count[num]++;
        }
        int n = nums.size();
        vector<vector<int>> buckets(n + 1);
        for (auto& [num, freq] : count) {
            buckets[freq].push_back(num);
        }
        vector<int> res;
        for (int i = n; i >= 0; i--) {
            for (int num : buckets[i]) {
                res.push_back(num);
                if (res.size() == k) {
                    return res;
                }
            }
        }
        return res;
    }
};

class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        unordered_map<int, int> count;
        for (int num : nums) {
            count[num]++;
        }
        priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
        for (auto& [num, freq] : count) {
            pq.push({freq, num}); 
            if (pq.size() > k) {
                pq.pop(); 
            }
        }
        vector<int> res;
        while (!pq.empty()) {
            res.push_back(pq.top().second);
            pq.pop();
        }
        // 如果题目要求“必须按频率从高到低输出”，因为小顶堆最先 pop 出来的是频率相对最小的，
        // 你需要在这里加一句：reverse(res.begin(), res.end());
        return res;
    }
};

# - vector> buckets(n + 1);
# - for (auto& [num, freq] : count) {
# - buckets[freq].pushback(num);
# - }

# - 学习一个叫做桶排序的东西。
# - 首先遍历数组做哈希表，数字和次数。

    for num in nums:
        count[num] = count.get(num, 0) + 1

    n = len(nums)
    buckets = [[] for _ in range(n + 1)]  # buckets[freq] 存所有出现 freq 次的数。
    for num, freq in count.items():
        buckets[freq].append(num)

    res = []
    for freq in range(n, 0 - 1, -1):
        for num in buckets[freq]:
            res.append(num)
            if len(res) == k:
                return res

    return res

5.3 自定义权重 Top K 比较器

自定义权重 Top K 比较器

题目简述

自定义权重 Top K：给定带分数和时间戳的候选项，按指定优先级选出前 K 项。

题解代码C++展开
// 关键注释
// 核心思路：堆顶始终是当前 Top K 中最差的候选。
// 关键关注：小分数更差。
// 易错提醒：固定大小堆中弹出的是“当前 Top K 中最差”的候选，这个语义必须和比较器一致。
struct Candidate {
    int value;
    int score;
    int timestamp;
};

struct WorseCandidateFirst {
    bool operator()(const Candidate& a, const Candidate& b) const {
        if (a.score != b.score) {
            return a.score > b.score; // 小分数更差；这里返回 true 让高分优先级更高。
        }
        return a.timestamp < b.timestamp; // 分数相同，时间更早者更差，优先被淘汰。
    }
};

vector<int> topKByCustomWeightTemplate(const vector<Candidate>& items, int k) {
    priority_queue<Candidate, vector<Candidate>, WorseCandidateFirst> heap;

    for (const Candidate& item : items) {
        heap.push(item);
        if ((int)heap.size() > k) {
            heap.pop(); // 堆顶始终是当前 Top K 中最差的候选。
        }
    }

    vector<int> ans;
    while (!heap.empty()) {
        ans.push_back(heap.top().value);
        heap.pop();
    }
    reverse(ans.begin(), ans.end()); // 如需从强到弱输出，可按业务规则再排序。
    return ans;
}
# 关键注释
# 核心思路：堆顶始终是当前 Top K 中最差的候选。
# 关键关注：小分数更差。
# 易错提醒：固定大小堆中弹出的是“当前 Top K 中最差”的候选，这个语义必须和比较器一致。
import heapq
from dataclasses import dataclass


@dataclass
class Candidate:
    value: int
    score: int
    timestamp: int


def topKByCustomWeightTemplate(items, k):
    heap = []
    for item in items:
        entry = (item.score, -item.timestamp, item.value, item)
        # 关键步骤：堆顶始终是当前 Top K 中最差的候选。
        heapq.heappush(heap, entry)
        if len(heap) > k:
            heapq.heappop(heap)

    ans = []
    while heap:
        # 边界提醒：固定大小堆中弹出的是“当前 Top K 中最差”的候选，这个语义必须和比较器一致。
        ans.append(heapq.heappop(heap)[3].value)
    ans.reverse()
    return ans

5.4 K 路归并小顶堆

23 合并 K 个升序链表

力扣原题 ↗
题目简述

23 合并 K 个升序链表：给定 K 个升序链表，合并全部结点并返回一个升序链表。

题解代码C++展开
// 关键注释（来自个人笔记）
// 思路：
// - 较容易想到的思路就是两两合并，在此基础上优化。第一条路是递归，即不断一分为二合并即可。
// - 另一条路就是两两合并、四四合并、八八合并···，重点关注这个双重for循环怎么写就行。
// - 或者使用优先队列（通过最小堆实现的），需要自己定义比较规则，语法见要点。实现优先队列后，pq顶上的就是最小值。所以遍历数组把每个链表的头节点放进去，顶上的就是所有链表里最小的，然后一个一个pop，每pop一个就把它的下一个push进去，循环完即可。
// 要点：
// - 优先队列语法为：
// - priorityqueue, CompareNode> pq;
// - 尖括号里：
// - 比较规则为：
// - struct CompareNode {
// - bool operator()(ListNode a, ListNode b) {
// - return a->val > b->val;
// - }
// - };
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* mergeTwoLists(ListNode* list1, ListNode* list2) {
        ListNode dummy; 
        ListNode* cur = &dummy; 
        while (list1 && list2) {
            if (list1->val < list2->val) {
                cur->next = list1; 
                list1 = list1->next;
            } else { 
                cur->next = list2; 
                list2 = list2->next;
            }
            cur = cur->next;
        }
        cur->next = list1 ? list1 : list2; 
        return dummy.next;
    }
    ListNode* mergeKLists(vector<ListNode*>& lists) {
        int m=lists.size();
        if (!m) return nullptr;
        for(int step=1;step<m;step*=2){
            for(int i=0;i<m-step;i+=2*step){
            lists[i]=mergeTwoLists(lists[i],lists[i+step]);
            }
        }
        return lists[0];
    }
};

/**
 * Definition for singly-linked list.
 * struct ListNode {
 * int val;
 * ListNode *next;
 * ListNode() : val(0), next(nullptr) {}
 * ListNode(int x) : val(x), next(nullptr) {}
 * ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */

// 👑 独立模块：金牌管家的“入场筛选规则”（仿函数）
struct CompareNode {
    // 重载括号运算符。注意：在 C++ 优先队列中，想要"最小堆"，这里必须用大于号 '>'
    bool operator()(ListNode* a, ListNode* b) {
        return a->val > b->val; 
    }
};

class Solution {
public:
    ListNode* mergeKLists(vector<ListNode*>& lists) {
        // 1. 召唤金牌管家。尖括号里清清爽爽：<数据类型, 容器类型, 比较规则>
        priority_queue<ListNode*, vector<ListNode*>, CompareNode> pq;

        // 2. 请所有班级的排头兵入场（遍历所有链表的头节点）
        for (ListNode* head : lists) {
            if (head != nullptr) {
                pq.push(head); 
            }
        }

        // 3. 准备接收最终队伍
        ListNode dummy(-1); 
        ListNode* cur = &dummy;

        // 4. 提取与替补的无限循环
        while (!pq.empty()) { 
            // 动作A：管家把最矮的人推到门口，你把他领走
            ListNode* node = pq.top(); 
            pq.pop();
            
            // 动作B：替补入场（把你领走的人的下一个小弟送进堆里）
            if (node->next != nullptr) { 
                pq.push(node->next); 
            }
            
            // 动作C：把领走的人接在最终的队伍后面
            cur->next = node; 
            cur = cur->next; 
        }

        return dummy.next; 
    }
};

# 思路：
# - 较容易想到的思路就是两两合并，在此基础上优化。第一条路是递归，即不断一分为二合并即可。
# - 另一条路就是两两合并、四四合并、八八合并···，重点关注这个双重for循环怎么写就行。
# - 或者使用优先队列（通过最小堆实现的），需要自己定义比较规则，语法见要点。实现优先队列后，pq顶上的就是最小值。所以遍历数组把每个链表的头节点放进去，顶上的就是所有链表里最小的，然后一个一个pop，每pop一个就把它的下一个push进去，循环完即可。
# 要点：
# - 优先队列语法为：
# - priorityqueue, CompareNode> pq;
# - 尖括号里：
# - 比较规则为：
# - struct CompareNode {
# - bool operator()(ListNode a, ListNode b) {
# - return a->val > b->val;
# - }
# - };
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

def mergeTwoListsTemplate(a, b):
    dummy = ListNode(0)
    cur = dummy

    while a and b:
        if a.val <= b.val:
            cur.next = a
            a = a.next
        else:
            cur.next = b
            b = b.next
        cur = cur.next

    cur.next = a if a else b
    return dummy.next

def mergeKByDivideTemplate(lists):
    if not lists:
        return None

    n = len(lists)
    step = 1

    # 每轮把间隔 step 的两条链表合并，step 翻倍。
    while step < n:
        for i in range(0, n - step, step * 2):
            lists[i] = mergeTwoListsTemplate(lists[i], lists[i + step])
        step *= 2

    return lists[0]

239 滑动窗口最大值：给定整数数组和窗口大小 K，返回窗口从左向右滑动时每个位置的最大值。

题解代码C++展开
// 关键注释（来自个人笔记）
// 思路：
// - 得想到双端队列呀，队列里存的一定是下标
// - 遍历循环里先进行踢人判断，如果不如新的值或者超标，就踢人
// - 超标：dq.front()=dq.back()
// - 记得判断队列是否为空
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        vector<int> res;
        deque<int> dq;
        for(int i=0;i<nums.size();i++){
           while(!dq.empty()&&dq.front()<i-k+1){
            dq.pop_front();
           }
           if(!dq.empty()&&nums[i]>=nums[dq.back()]){
            dq.pop_back();
           }
           dq.push_back(i);
           if(i-k+1>=0) res.push_back(nums[dq.front()]);
        }
        return res;
    }
};
# 关键注释（来自个人笔记）
# 思路：
# - 得想到双端队列呀，队列里存的一定是下标
# - 遍历循环里先进行踢人判断，如果不如新的值或者超标，就踢人
# - 超标：dq.front()=dq.back()
# - 记得判断队列是否为空
def maxSlidingWindow(nums, k):
    res = []
    dq = deque()  # 存下标；保持 nums[dq[0]] >= nums[dq[1]] >= ...。

    for i in range(0, len(nums)):
        # 1. 踢过期队头：当前窗口是 [i - k + 1, i]。
        while not not dq and dq[0] < i - k + 1:
            dq.popleft()

        # 2. 踢弱队尾：nums[i] 更靠右且不小于队尾，队尾永远不会再成为最大值。
        # 必须用 while，因为可能有多个队尾候选都不如当前值。
        while not not dq and nums[i] >= nums[dq[-1]]:
            dq.pop()

        # 3. 当前下标入队，成为未来窗口的候选最大值。
        dq.append(i)

        # 4. 窗口长度达到 k 后，队头就是当前窗口最大值。
        if i - k + 1 >= 0:
            res.append(nums[dq[0]])

    return res

class MedianFinder {
private:
    priority_queue<int, vector<int>, less<int>> maxHeap;
    priority_queue<int, vector<int>, greater<int>> minHeap;

public:
    MedianFinder() {
    }
    
    void addNum(int num) {
        maxHeap.push(num);
        minHeap.push(maxHeap.top());
        maxHeap.pop();
        if (minHeap.size() > maxHeap.size()) {
            maxHeap.push(minHeap.top());
            minHeap.pop();
        }
    }
    
    double findMedian() {
        if (maxHeap.size() > minHeap.size()) {
            return maxHeap.top();
        }
        return (maxHeap.top() + minHeap.top()) / 2.0;
    }
};

import heapq


class MedianFinder:
    def __init__(self):
        self.maxHeap = []
        self.minHeap = []

    def addNum(self, num):
        heapq.heappush(self.maxHeap, -num)
        heapq.heappush(self.minHeap, -heapq.heappop(self.maxHeap))
        if len(self.minHeap) > len(self.maxHeap):
            heapq.heappush(self.maxHeap, -heapq.heappop(self.minHeap))

    def findMedian(self):
        if len(self.maxHeap) > len(self.minHeap):
            return float(-self.maxHeap[0])
        return (-self.maxHeap[0] + self.minHeap[0]) / 2.0
